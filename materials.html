Quickstart

Replace with the real commands once the repo is final

bash
# 1. Build the workspace (ROS 2 Humble)
mkdir -p ~/fetchbot_ws/src && cd ~/fetchbot_ws/src
git clone https://github.com/abelblue3/fetchbot.git .
cd .. && rosdep install --from-paths src -y --ignore-src
colcon build --symlink-install && source install/setup.bash

# 2. On the robot: base, LiDAR, camera and arm driver
ros2 launch fetchbot_bringup robot.launch.py

# 3. On the laptop: localization, Nav2, MoveIt, perception and mission
ros2 launch fetchbot_bringup fetch.launch.py map:=maps/lab.yaml
